//  aob_l2_arrays.inc — 跨构建多层特征码（本文件由工具生成, 请勿手改）
//  ---------------------------------------------------------------------
//  生成: python Tools/gen_aob_relaxed.py
//  校验: 四构建 (sr_hv.exe / sr_hv_gog.exe / sr_hv_epic.exe / MS Store 裸段)
//        每个 hook 都必须能唯一命中其对应层, 否则生成器拒绝出表。
//
//  L2  R2_*/RM_*  —— L1 落空时的第一次放宽。通配规则由反汇编机械推导:
//        基址 rsp/rbp/r11 的 disp(栈帧溢出槽) / 基址 rip 的 disp(全局地址)
//        / `sub|add rsp, imm`(帧大小) / `mov reg, imm32` + 紧跟 call
//        (__chkstk 帧大小) / `call|jmp rel`(代码地址)。
//        保留: 操作码/ModRM/SIB/寄存器编码、语义立即数(cmp/bt/shl)、
//              结构体字段偏移、函数内条件短跳转。
//  教训: 手工圈通配区间会漏。Format 窗口内第二个栈帧相对 disp32
//        `mov [rbp+2F50h],rax` 曾被漏掉, 而它恰随帧大小漂移。
//
//  L3  R3_*  —— **换代构建**的精确特征码, 纯精确(无 mask)。
//        Microsoft Store 版 (sriv.exe, TDS=5E58CEF8, 2020-02) 的
//        Format / Subtitle 换了一代 MSVC, 前导形态整个不同:
//          新: push rbp/si/di/r15; lea rbp,[rsp-2F88h]; mov eax,3088h;
//              call __chkstk; sub rsp,rax
//          老: mov [rsp+20h],r9; mov [rsp+18h],r8; mov [rsp+8],rcx  ← 参数
//              push rbp/rbx/rsi/r14; lea rbp,[rsp-2F58h]            home 区
//              mov eax,3058h; call __chkstk; sub rsp,rax             先落地
//        指令长度都不同, 任何字节 mask 都救不了 —— 只能构建族各一套。
//        同一构建内 RIP 相对位移与 rel32 不随 ASLR 变化, 故无需 mask。
//  详见 Tools/aob_relaxed_report.txt。
// =====================================================================
